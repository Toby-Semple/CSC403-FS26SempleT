    \item ~\\

        \begin{enumerate}

            \addtocounter{enumi}{1}

            \item ~\\
                Premises: \\
                 - The main course will be either beef or fish.\\
                 - The vegetable will be either peas or corn.\\
                 - We will not have both fish as the main course and corn as the vegetable. \\
                ~\\
                Conclusion: \\
                 - We will not have both beef as the main course and peas as the vegetable. \\
                ~\\
                This conclusion is incorrect, because having both beef as the main course and peas as the vegetable still satisfies all of the premises.

            \item ~\\
                Premises: \\
                 - Either John or Bill is telling the truth.\\
                 - Either Bill or Sam is lying.
                ~\\
                Conclusion: \\
                 - Either John is telling the truth or Sam is lying. \\
                ~\\
                This conclusion is correct:\\
                 - If John is lying, then Bill must be telling the truth. If Bill is telling the truth, then Bill isn't lying, which means Sam is. \\
                 - If Sam is telling the truth, then Bill must be lying. If Bill is lying, then he isn't telling the truth, which means John is. 
                

        \end{enumerate}

    
\end{enumerate}
